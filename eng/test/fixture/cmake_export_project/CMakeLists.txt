function(argyle_fixture_assert_scalar_copy)
    get_property(_consumer GLOBAL PROPERTY ARGYLE_FIXTURE_CONSUMER_TARGET)
    get_property(_probe GLOBAL PROPERTY ARGYLE_NIMBLE_CONTEXT_PROBE_TARGET)
    if(NOT _consumer OR NOT _probe)
        message(FATAL_ERROR "fixture scalar-property assertion has incomplete target state")
    endif()

    set(_report "${CMAKE_BINARY_DIR}/argyle-probe-refresh.txt")
    file(WRITE "${_report}" "")
    foreach(_property IN ITEMS C_STANDARD C_STANDARD_REQUIRED C_EXTENSIONS
            POSITION_INDEPENDENT_CODE INTERPROCEDURAL_OPTIMIZATION_RELEASE)
        get_property(_consumer_set TARGET "${_consumer}" PROPERTY "${_property}" SET)
        get_property(_probe_set TARGET "${_probe}" PROPERTY "${_property}" SET)
        get_target_property(_consumer_value "${_consumer}" "${_property}")
        get_target_property(_probe_value "${_probe}" "${_property}")
        if(NOT _consumer_set OR NOT _probe_set OR NOT "${_consumer_value}" STREQUAL "${_probe_value}")
            message(FATAL_ERROR "late ${_property} was not refreshed onto the CMake exporter probe")
        endif()
        file(APPEND "${_report}" "${_property}=${_probe_value}\n")
    endforeach()

    get_target_property(_probe_links "${_probe}" LINK_LIBRARIES)
    list(FIND _probe_links "${_consumer}" _consumer_link_index)
    if(NOT _consumer_link_index EQUAL -1)
        message(FATAL_ERROR "exporter probe must not link the consumer target")
    endif()
    file(APPEND "${_report}" "probe_links_consumer=OFF\n")
endfunction()
